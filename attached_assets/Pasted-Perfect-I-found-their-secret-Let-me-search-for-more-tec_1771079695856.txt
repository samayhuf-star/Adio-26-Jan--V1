async function generateKeywordsAlphabetSoup(seedKeyword) {
  const keywords = new Set();
  const alphabet = 'abcdefghijklmnopqrstuvwxyz'.split('');
  
  // Pattern 1: Append A-Z
  for (const letter of alphabet) {
    const query = `${seedKeyword} ${letter}`;
    const suggestions = await getGoogleAutocomplete(query);
    suggestions.forEach(kw => keywords.add(kw));
  }
  
  // Pattern 2: Prepend A-Z
  for (const letter of alphabet) {
    const query = `${letter} ${seedKeyword}`;
    const suggestions = await getGoogleAutocomplete(query);
    suggestions.forEach(kw => keywords.add(kw));
  }
  
  // Pattern 3: Append 0-9
  for (let num = 0; num <= 9; num++) {
    const query = `${seedKeyword} ${num}`;
    const suggestions = await getGoogleAutocomplete(query);
    suggestions.forEach(kw => keywords.add(kw));
  }
  
  // Pattern 4: Between words (for multi-word seeds)
  const words = seedKeyword.split(' ');
  if (words.length > 1) {
    for (const letter of alphabet) {
      const query = `${words[0]} ${letter} ${words.slice(1).join(' ')}`;
      const suggestions = await getGoogleAutocomplete(query);
      suggestions.forEach(kw => keywords.add(kw));
    }
  }
  
  return Array.from(keywords);
}

// Google Autocomplete API
async function getGoogleAutocomplete(query) {
  const url = `https://www.google.com/complete/search?q=${encodeURIComponent(query)}&client=chrome`;
  const response = await fetch(url);
  const data = await response.json();
  return data[1] || []; // Returns array of suggestions
}
