## 📊 INTELLIGENT VALIDATION

### **Auto-Filter Bad Keywords:**

```javascript
function validateKeywords(keywords, context) {
  const industryTerms = {
    'travel': ['flight', 'airline', 'travel', 'booking', 'ticket'],
    'plumbing': ['plumber', 'pipe', 'leak', 'drain'],
    'hvac': ['ac', 'hvac', 'heating', 'cooling']
  };
  
  const requiredTerms = industryTerms[context.industry];
  
  return keywords.filter(kw => {
    const keyword = kw.keyword.toLowerCase();
    
    // Must have industry term
    const hasIndustryTerm = requiredTerms.some(term => 
      keyword.includes(term)
    );
    
    // Must NOT have wrong industry terms
    const wrongIndustries = Object.keys(industryTerms)
      .filter(ind => ind !== context.industry);
    
    const hasWrongTerm = wrongIndustries.some(wrongInd =>
      industryTerms[wrongInd].some(term => keyword.includes(term))
    );
    
    // Validation
    if (!hasIndustryTerm) {
      console.log(`❌ Rejected: ${kw.keyword} (no industry term)`);
      return false;
    }
    
    if (hasWrongTerm) {
      console.log(`❌ Rejected: ${kw.keyword} (wrong industry)`);
      return false;
    }
    
    if (kw.relevanceScore < 75) {
      console.log(`❌ Rejected: ${kw.keyword} (score: ${kw.relevanceScore})`);
      return false;
    }
    
    return true;
  });
}
```
